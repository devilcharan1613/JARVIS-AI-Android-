cmake_minimum_required(VERSION 3.22.1)
project(jarvis_llama)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp/CMakeLists.txt")
 message(FATAL_ERROR "llama.cpp source missing. Run scripts/fetch-llama.sh first.")
endif()
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
add_subdirectory(llama.cpp)
add_library(jarvis_llama SHARED native-lib.cpp)
target_include_directories(jarvis_llama PRIVATE llama.cpp/include)
find_library(log-lib log)
target_link_libraries(jarvis_llama PRIVATE llama ${log-lib})
